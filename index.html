// --- Riyadh Premier League Season 25 Sync Script ---
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "No data received" }))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "No active spreadsheet" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // 1. Teams Sheet
    var sheetTeams = getOrCreateSheet(ss, "Teams");
    sheetTeams.clearContents();
    sheetTeams.appendRow(["Team ID", "Franchise Name", "Code", "Pool", "Color"]);
    sheetTeams.getRange(1, 1, 1, 5).setFontWeight("bold").setBackground("#F59E0B").setFontColor("#000000");
    if (data.teams && data.teams.length > 0) {
      data.teams.forEach(function(t) {
        sheetTeams.appendRow([t.id, t.name, t.code, t.group || "A", t.color]);
      });
    }

    // 2. Fixtures Sheet
    var sheetFix = getOrCreateSheet(ss, "Fixtures");
    sheetFix.clearContents();
    sheetFix.appendRow(["Match #", "Round", "Date", "Time", "Venue", "Pool", "Home Franchise", "Score A", "Away Franchise", "Score B", "Result / Winner"]);
    sheetFix.getRange(1, 1, 1, 11).setFontWeight("bold").setBackground("#10B981").setFontColor("#FFFFFF");
    if (data.fixtures && data.fixtures.length > 0) {
      data.fixtures.forEach(function(r) {
        r.matches.forEach(function(m) {
          sheetFix.appendRow([
            m.matchNumber,
            "Round " + m.round,
            m.date,
            m.time,
            m.venue,
            m.group ? "Pool " + m.group : "League",
            m.homeTeam ? m.homeTeam.name : "",
            m.scoreA || "-",
            m.awayTeam ? m.awayTeam.name : "",
            m.scoreB || "-",
            m.summary || (m.winner ? "Winner: " + m.winner : "Upcoming")
          ]);
        });
      });
    }

    // 3. Points Table Sheet
    var sheetPts = getOrCreateSheet(ss, "Points_Table");
    sheetPts.clearContents();
    sheetPts.appendRow(["Franchise", "Pool", "Played", "Won", "Lost", "Points"]);
    sheetPts.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#3B82F6").setFontColor("#FFFFFF");
    if (data.standings && data.standings.length > 0) {
      data.standings.forEach(function(s) {
        sheetPts.appendRow([s.teamName, "Pool " + (s.group || "A"), s.played, s.won, s.lost, s.pts]);
      });
    }

    // 4. Knockouts Sheet
    if (data.knockouts) {
      var sheetKO = getOrCreateSheet(ss, "Knockouts");
      sheetKO.clearContents();
      sheetKO.appendRow(["Stage", "Date", "Venue", "Team A", "Team B", "Winner"]);
      sheetKO.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#8B5CF6").setFontColor("#FFFFFF");
      var koKeys = ["q1", "elim", "q2", "final"];
      koKeys.forEach(function(k) {
        var match = data.knockouts[k];
        if (match) {
          sheetKO.appendRow([match.title, match.date, match.venue, match.teamA, match.teamB, match.winner]);
        }
      });
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success", timestamp: new Date() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ status: "online", tournament: "RPL Season 25" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) { sheet = ss.insertSheet(name); }
  return sheet;
}
